package com.example.uyg13ss254;

import androidx.appcompat.app.AppCompatActivity;

import android.content.Intent;
import android.os.Bundle;
import android.view.View;

import com.example.uyg13ss254.databinding.ActivityMainBinding;

public class MainActivity extends AppCompatActivity {

    private ActivityMainBinding binding;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        binding = ActivityMainBinding.inflate(getLayoutInflater());
        View view = binding.getRoot();
        setContentView(view);
    }

    public void digerActivity(View view){

        String adiSoyadi = binding.editTextTextAdiSoyadi.getText().toString();
        String telefonNo = binding.editTextTelefon.getText().toString();

        Intent intent = new Intent(MainActivity.this, digerActivity.class);

        intent.putExtra("adiSoyadiKey", adiSoyadi);
        intent.putExtra("telefonNoKey", telefonNo);

        startActivity(intent);
    }
}

package com.example.uyg13ss254;

import androidx.appcompat.app.AppCompatActivity;

import android.content.Intent;
import android.os.Bundle;

import com.example.uyg13ss254.databinding.ActivityDigerBinding;

public class digerActivity extends AppCompatActivity {

    private ActivityDigerBinding binding;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        binding = ActivityDigerBinding.inflate(getLayoutInflater());
        setContentView(binding.getRoot());

        Intent intent = getIntent();

        String adiSoyadi = intent.getStringExtra("adiSoyadiKey");
        String telefonNo = intent.getStringExtra("telefonNoKey");

        binding.textViewAdiSoyadi.setText(adiSoyadi);
        binding.textViewTelefonNo.setText(telefonNo);
    }
}
