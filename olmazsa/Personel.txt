package com.example.personel193;


import androidx.appcompat.app.AppCompatActivity;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;

public class MainActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {

        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        EditText editTextAd = findViewById(R.id.editTextAd);
        EditText editTextMaas = findViewById(R.id.editTextMaas);
        Button buttonKaydet = findViewById(R.id.buttonKaydet);
        TextView textViewSonuc = findViewById(R.id.textViewSonuc);

        buttonKaydet.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View view) {

                String ad = editTextAd.getText().toString();
                double maas = Double.parseDouble(editTextMaas.getText().toString());

                Personel personel = new Personel();

                personel.setAd(ad);
                personel.setMaas(maas);

                String sonuc = "Ad: " + personel.getAd() +
                        "\nMaaş: " + personel.getMaas();

                textViewSonuc.setText(sonuc);
            }
        });
    }
}

package com.example.personel193;



import android.util.Log;

public class Personel {

    private String ad;
    private double maas;   // Kapsülleme (private)


    public String getAd() {
        return ad;
    }

    public void setAd(String ad) {
        this.ad = ad;
    }


    public double getMaas() {
        return maas;
    }

    public void setMaas(double maas) {

        if (maas < 4253) {
            Log.e("Kapsülleme", "Maaş 4253'ten küçük olamaz!");
            this.maas = 4253;
        } else {
            this.maas = maas;
        }
    }
}
