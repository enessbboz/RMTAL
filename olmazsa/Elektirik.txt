package com.example.a181ss;

import androidx.appcompat.app.AppCompatActivity;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;

public class MainActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        Button btnHesapla = findViewById(R.id.buttonHesapla);

        btnHesapla.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View view) {

                EditText editTextCihazAdi = findViewById(R.id.editTextCihazAdi);
                EditText editTextKW = findViewById(R.id.editTextKw);
                EditText editTextSaat = findViewById(R.id.editTextGunlukKullanimSaat);
                TextView textViewSonuc = findViewById(R.id.textViewSonuc);

                String cihazAdi = editTextCihazAdi.getText().toString();
                double kw = Double.parseDouble(editTextKW.getText().toString());
                double saat = Double.parseDouble(editTextSaat.getText().toString());

                ElektrikliCihaz cihaz = new ElektrikliCihaz();
                cihaz.cihazAdi = cihazAdi;
                cihaz.cihazKW = kw;
                cihaz.gunlukSaatKullanim = saat;

                double tuketim = cihaz.aylikTuketim();
                double ucret = cihaz.aylikUcret();

                textViewSonuc.setText(
                        "Cihaz: " + cihazAdi +
                                "\nAylık Tüketim: " + tuketim + " kWh" +
                                "\nAylık Ücret: " + ucret + " TL"
                );
            }
        });
    }
}

package com.example.a181ss;
public class ElektrikliCihaz {

    public String cihazAdi;
    public double cihazKW;
    public double gunlukSaatKullanim;


    public double aylikTuketim() {
        return cihazKW * gunlukSaatKullanim * 30;
    }


    public double aylikUcret() {
        return aylikTuketim() * 1.37;
    }
}
